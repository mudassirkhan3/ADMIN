var SHEET_NAME = "Sheet1"; function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var doc = SpreadsheetApp.getActiveSpreadsheet(); var sheet = doc.getSheetByName(SHEET_NAME); // Auto-setup exact headers if the sheet is empty if (sheet.getLastRow() === 0) { var headers = [ "Timestamp", "fullname", "fathername", "cnic", "dob", "district", "temporary_address", "permanent_address", "phone", "emergency_contact", "matric_marks", "fsc_marks", "candidate_photo", "agreement_accepted", "payment_id", "Status", "Payment_Status" ]; sheet.getRange(1, 1, 1, headers.length).setValues([headers]); } var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; var nextRow = sheet.getLastRow() + 1; var rowData = headers.map(function(header) { if (header === 'Timestamp') { return new Date(); } if (header === 'Status') { return 'Under Review'; } if (header === 'Payment_Status') { return 'Pending Verification'; } // Maps html form field 'payment_id' (or fallback alternatives) if (header === 'payment_id') { return e.parameter['payment_id'] || e.parameter['payment_ref'] || ''; } return e.parameter[header] || ''; }); sheet.getRange(nextRow, 1, 1, rowData.length).setValues([rowData]); return ContentService .createTextOutput(JSON.stringify({ 'result': 'success', 'row': nextRow })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ 'result': 'error', 'error': err.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } function doGet(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var doc = SpreadsheetApp.getActiveSpreadsheet(); var sheet = doc.getSheetByName(SHEET_NAME); var rows = sheet.getDataRange().getValues(); if (rows.length === 0) { return ContentService .createTextOutput(JSON.stringify({ 'status': 'error', 'message': 'Sheet is empty' })) .setMimeType(ContentService.MimeType.JSON); } var headers = rows[0]; var query = e.parameter.q; // Handle admin stats request if (query === 'admin_stats') { var total = rows.length - 1; var approved = 0; var pending = 0; var statusIdx = headers.indexOf('Status'); for (var i = 1; i < rows.length; i++) { var status = rows[i][statusIdx] ? rows[i][statusIdx].toString().toLowerCase() : ''; if (status.includes('approved') || status.includes('verified')) { approved++; } else { pending++; } } return ContentService .createTextOutput(JSON.stringify({ 'total': total, 'approved': approved, 'pending': pending })) .setMimeType(ContentService.MimeType.JSON); } if (!query) { return ContentService .createTextOutput(JSON.stringify({ 'status': 'error', 'message': 'No query provided' })) .setMimeType(ContentService.MimeType.JSON); } var cnicIdx = headers.indexOf('cnic'); var phoneIdx = headers.indexOf('phone'); var nameIdx = headers.indexOf('fullname'); var fatherIdx = headers.indexOf('fathername'); var statusIdx = headers.indexOf('Status'); var payStatusIdx = headers.indexOf('Payment_Status'); var photoIdx = headers.indexOf('candidate_photo'); for (var i = 1; i < rows.length; i++) { var row = rows[i]; var rowCnic = row[cnicIdx] ? row[cnicIdx].toString().trim() : ''; var rowPhone = row[phoneIdx] ? row[phoneIdx].toString().trim() : ''; if (rowCnic.toLowerCase() === query.toLowerCase() || rowPhone.toLowerCase() === query.toLowerCase()) { return ContentService .createTextOutput(JSON.stringify({ 'status': 'success', 'name': row[nameIdx], 'father': row[fatherIdx], 'cnic': rowCnic, 'phone': rowPhone, 'appStatus': row[statusIdx] || 'Under Review', 'payStatus': row[payStatusIdx] || 'Pending Verification', 'photo': row[photoIdx] || '' })) .setMimeType(ContentService.MimeType.JSON); } } return ContentService .createTextOutput(JSON.stringify({ 'status': 'not_found' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ 'status': 'error', 'message': err.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
